337 // Функция скомпилируется только для тех типов, для которых HasIntrospection<T>::result будет true
338 // Пример:
339 //
340 // class A {};
341 // class B : public class A { ... INTROSPECTION ... }
342 // B *b;
343 // GetIntrospection(b) // вернет интроспекцию класса B
344 // GetIntrospection((A *) b) // вернет nullptr, т.к. будет вызвана функция #4, см. ниже
345 //

348 {
349  const InspInfo* ret = nullptr;
350

353  ret = t->GetTypeInfo();
354  }
355 
356  return ret;
357 }
