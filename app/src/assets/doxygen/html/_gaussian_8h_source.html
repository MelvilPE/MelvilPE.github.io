35  float32 k2 = -1.f / (2 * q * q);
36 
37  Vector<float32> weights(halfKernelSize);

39 
40  float32 step = 1.f / float32(halfKernelSize - 1);
41  for (int32 i = 1; i <= halfKernelSize; ++i)
42  {
43  weights[halfKernelSize - i] = k1 * exp(k2 * x * x);
44  x += step;
45  }

59  }
