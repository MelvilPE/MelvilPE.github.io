145 {
146  FastName stdName(UTF8Utils::EncodeToUTF8(name));
147 
148  auto stdIter = TimeZoneHelper::winStdToCommonNamesMap.find(stdName);
149  if (stdIter != TimeZoneHelper::winStdToCommonNamesMap.end())
150  {
151  stdName = TimeZoneHelper::winStdToCommonNamesMap.at(stdName);
152  }
153 
154  auto iter = TimeZoneHelper::namesMap.find(stdName);
155  if (iter == TimeZoneHelper::namesMap.end())
156  {
157  return String("");
158  }
159  FastName commonName = iter->second;
