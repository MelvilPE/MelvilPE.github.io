164 inline uint64_t HashCombine64(uint64_t hash, uint64_t combined)
165 {
166  uint32_t hL = static_cast<uint32_t>(hash & 0xFFFFFFFF);
167  uint32_t hH = static_cast<uint32_t>(hash >> 32);
168  uint32_t cL = static_cast<uint32_t>(combined & 0xFFFFFFFF);
169  uint32_t cH = static_cast<uint32_t>(combined >> 32);
170 
171  uint32_t ebx = cL;
172  uint32_t edi = cH;
173 
174  uint32_t t_edi = (edi << 26) | (ebx >> 6);
175  uint32_t t_ebx = (ebx << 26);
176  ebx = t_ebx;
177  edi = t_edi;
178 
179  uint64_t sum1 = static_cast<uint64_t>(ebx) + hH;
180  ebx = static_cast<uint32_t>(sum1 & 0xFFFFFFFF);
181  edi = edi + hL + static_cast<uint32_t>(sum1 >> 32);
182 
183  uint32_t t_edi2 = (edi << 6) | (ebx >> 26);
184  uint32_t t_ebx2 = (ebx << 6);
185  ebx = t_ebx2;
186  edi = t_edi2;
187 
188  uint32_t partL = (hH >> 2) | (hL << 30);
189  uint32_t partH = hL >> 2;
190  uint64_t sum2 = static_cast<uint64_t>(ebx) + partL;
191  ebx = static_cast<uint32_t>(sum2 & 0xFFFFFFFF);
192  edi = edi + partH + static_cast<uint32_t>(sum2 >> 32);
193 
194  uint64_t sum3 = static_cast<uint64_t>(ebx) + 0x9E3779B9;
195  ebx = static_cast<uint32_t>(sum3 & 0xFFFFFFFF);
196  edi = edi + 0x9E3779B9 + static_cast<uint32_t>(sum3 >> 32);
197 
198  ebx ^= cL;
199  edi ^= cH;
200 
201  uint32_t resH = hH ^ ebx;
202  uint32_t resL = hL ^ edi;
203 
204  return (static_cast<uint64_t>(resH) << 32) | resL;
205 }

216 {

219  auto hash_mix = [](uint32& a, uint32& b, uint32& c)
220  {
221  a -= b;
222  a -= c;
223  a ^= (c >> 13);
224  b -= c;
225  b -= a;
226  b ^= (a << 8);
227  c -= a;
228  c -= b;
229  c ^= (b >> 13);
230  a -= b;
231  a -= c;
232  a ^= (c >> 12);
233  b -= c;
234  b -= a;
235  b ^= (a << 16);
236  c -= a;
237  c -= b;
238  c ^= (b >> 5);
239  a -= b;
240  a -= c;
241  a ^= (c >> 3);
242  b -= c;
243  b -= a;
244  b ^= (a << 10);
245  c -= a;
246  c -= b;
247  c ^= (b >> 15);
248  };
249 
250  // set up the internal state
251 
252  uint32 len = length;
253  uint32 a = 0x9E3779B9; // the golden ratio, an arbitrary value
254  uint32 b = 0x9E3779B9; // the golden ratio, an arbitrary value
255  uint32 c = 0; // the previous hash value
256 
257  // handle most of the key
258 
259  while (len >= 12)
260  {
261  a += (key[0] + (static_cast<uint32>(key[1]) << 8) + (static_cast<uint32>(key[2]) << 16) + (static_cast<uint32>(key[3]) << 24));
262  b += (key[4] + (static_cast<uint32>(key[5]) << 8) + (static_cast<uint32>(key[6]) << 16) + (static_cast<uint32>(key[7]) << 24));
263  c += (key[8] + (static_cast<uint32>(key[9]) << 8) + (static_cast<uint32>(key[10]) << 16) + (static_cast<uint32>(key[11]) << 24));
264 
265  hash_mix(a, b, c);
266 
267  key += 12;
268  len -= 12;
269  }
270 
271  // handle the last 11 bytes
272 
273  c += length;
274 
275  //lint -e{616} // control flows into case/default
276  switch (len)
277  {
278  // all the case statements fall through
279 
280  case 11:
281  c += (static_cast<uint32>(key[10]) << 24);
282  DAVA_SWITCH_CASE_FALLTHROUGH;
283  case 10:
284  c += (static_cast<uint32>(key[9]) << 16);
285  DAVA_SWITCH_CASE_FALLTHROUGH;
286  case 9:
287  c += (static_cast<uint32>(key[8]) << 8);
288  DAVA_SWITCH_CASE_FALLTHROUGH;
289  // the first byte of c is reserved for the length
290  case 8:
291  b += (static_cast<uint32>(key[7]) << 24);
292  DAVA_SWITCH_CASE_FALLTHROUGH;
293  case 7:
294  b += (static_cast<uint32>(key[6]) << 16);
295  DAVA_SWITCH_CASE_FALLTHROUGH;
296  case 6:
297  b += (static_cast<uint32>(key[5]) << 8);
298  DAVA_SWITCH_CASE_FALLTHROUGH;
299  case 5:
300  b += key[4];
301  DAVA_SWITCH_CASE_FALLTHROUGH;
302  case 4:
303  a += (static_cast<uint32>(key[3]) << 24);
304  DAVA_SWITCH_CASE_FALLTHROUGH;
305  case 3:
306  a += (static_cast<uint32>(key[2]) << 16);
307  DAVA_SWITCH_CASE_FALLTHROUGH;
308  case 2:
309  a += (static_cast<uint32>(key[1]) << 8);
310  DAVA_SWITCH_CASE_FALLTHROUGH;
311  case 1:
312  a += key[0];
313  DAVA_SWITCH_CASE_FALLTHROUGH;
314  case 0:
315  // case 0: nothing left to add
316  break;
317  }
