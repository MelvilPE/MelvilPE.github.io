10 /*
11  template class ThreadLocalPtr - implementation of cross-platform thread local storage (TLS). Each instance of ThreadLocalPtr
12  represents a pointer to an object of type T. Each thread has a distinct value.
13 
14  To obtain value for the current thread one can use Get() method, or -> and * pointer dereference operators. Initially pointer
15  has nullptr value for each thread, to set value for the current thread one can use Reset() method.
16 
17  ThreadLocalPtr's interface is similar to interface of boost::thread_specific_ptr.
18 
19  C++ 11 supports thread_local keyword which does the same thing but not all compilers support it.
20  Also, compiler specific stuff (__declspec(thread), __thread) does not work well between platforms.
21 
22  Restrictions:
23  variables of type ThreadLocal can have only static storage duration (global or local static, and static data member)
24  if you declare ThreadLocal as automatic object it's your own problems, so don't cry: Houston, we've got a problem
25 
26  TODO:
27  integrate ThreadLocalPtr into DAVA::Thread to support automatic cleanup on thread exit. For now user is responsible for
28  calling ThreadLocalPtr::Reset() to delete pointer
29 */
30 template <typename T>
31 class ThreadLocalPtr final
32 {
33 #if defined(__DAVAENGINE_WINDOWS__)
34  using KeyType = DWORD;
35 #elif defined(__DAVAENGINE_POSIX__)
36  using KeyType = pthread_key_t;
37 #else
38 #error "ThreadLocalPtr: platform is unknown"
39 #endif
40 
41 public:
42  ThreadLocalPtr() DAVA_NOEXCEPT;
43  ThreadLocalPtr(void (*deleter_)(T*)) DAVA_NOEXCEPT;
44  ~ThreadLocalPtr() DAVA_NOEXCEPT;
45 
46  T* Get() const DAVA_NOEXCEPT;
47  T* operator->() const DAVA_NOEXCEPT;
48  T& operator*() const DAVA_NOEXCEPT;
49 
50  // Get current pointer and set nullptr without deleting
51  T* Release() DAVA_NOEXCEPT;
52  // Set new pointer, delete previous pointer if it is not same with new pointer
53  void Reset(T* newValue = nullptr) DAVA_NOEXCEPT;
54 
55  // Method to test whether thread local storage has been successfully created by system
56  bool IsCreated() const DAVA_NOEXCEPT;
57 
58 private:
59  ThreadLocalPtr(const ThreadLocalPtr&) = delete;
60  ThreadLocalPtr& operator=(const ThreadLocalPtr&) = delete;
61 
62  // Platform spefific methods
63  void CreateTlsKey() DAVA_NOEXCEPT;
64  void DeleteTlsKey() const DAVA_NOEXCEPT;
65  void SetTlsValue(void* rawValue) const DAVA_NOEXCEPT;
66  void* GetTlsValue() const DAVA_NOEXCEPT;
67 
68  static void DefaultDeleter(T* ptr) DAVA_NOEXCEPT;
69 
70 private:
71  KeyType key;
72  bool isCreated = false;
73  void (*deleter)(T*);
74 };
