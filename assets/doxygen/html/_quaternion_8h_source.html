87 inline void Quaternion::Construct(const Vector3& source, const Vector3& dest, const Vector3& fallbackAxis)
88 {
89  // Based on Stan Melax's article in Game Programming Gems
90  // Copy, since cannot modify local
91  Vector3 v0 = source;
92  Vector3 v1 = dest;
93  v0.Normalize();
94  v1.Normalize();
95 
96  float32 d = v0.DotProduct(v1);
97  // If dot == 1, vectors are the same
98  if (d >= 1.0f)
99  {
100  Set(); //Quaternion::IDENTITY;
101  }
102  if (d < (1e-6f - 1.0f))
103  {
104  if (fallbackAxis != Vector3(0, 0, 0))
105  {
106  // rotate 180 degrees about the fallback axis
107  Construct(fallbackAxis, PI);
108  }
109  else
110  {
111  // Generate an axis
112  Vector3 axisX(1, 0, 0);
113  Vector3 axis = axisX.CrossProduct(source);
114  if (axis.IsZero()) // pick another if colinear
115  {
116  Vector3 axisY(0, 1, 0);
117  axis = axisY.CrossProduct(source);
118  }
119  axis.Normalize();
120  Construct(axis, PI);
121  }
122  }
123  else
124  {
125  float32 s = std::sqrt((1 + d) * 2);
126  float32 invs = 1 / s;
127 
128  Vector3 c = v0.CrossProduct(v1);
129 
130  x = c.x * invs;
131  y = c.y * invs;
132  z = c.z * invs;
133  w = s * 0.5f;
134  Normalize();
135  }
136 }

215 DAVA::Vector3 Quaternion::GetEuler() const
216 {
217  /*
218  [ 1-2y2-2z2 2xy-2wz 2xz+2wy ]
219  [ 2xy+2wz 1-2x2-2z2 2yz-2wx ]
220  [ 2xz-2wy 2yz+2wx 1-2x2-2y2 ]
221  */
222 
223  // float32 wx, wy, wz, xx, yy, yz, xy, xz, zz;
224  // xx = x * x;
225  // xy = x * y;
226  // xz = x * z;
227  // yy = y * y;
228  // yz = y * z;
229  // zz = z * z;
230  // wx = w * x;
231  // wy = w * y;
232  // wz = w * z;
233 
234  Vector3 euler;
235  // float32 m_21 = 2.0f * (yz + wx);
236  // float32 m_22 = 1.0f - 2.0f * (xx + yy);
237  // float32 m_20 = 2.0f * (xz - wy);
238  // float32 m_10 = 2.0f * (xy + wz);
239  // float32 m_00 = 1.0f - 2.0f * (yy + zz);
240 
241  // euler.x = std::atan2(m_21, m_22);
242  // euler.y = std::atan2(-m_20, std::sqrt(m_21 * m_21 + m_22 * m_22));
243  // euler.z = std::atan2(m_10, m_00);
244 
245  float32 wx, wy, wz, xx, yy, yz, xy, xz, zz, x2, y2, z2;
246  x2 = x + x;
247  y2 = y + y;
248  z2 = z + z;
249  xx = x * x2;
250  xy = x * y2;
251  xz = x * z2;
252  yy = y * y2;
253  yz = y * z2;
254  zz = z * z2;
255  wx = w * x2;
256  wy = w * y2;
257  wz = w * z2;
258 
259  float32 m_12 = yz + wx;
260  float32 m_22 = 1.0f - (xx + yy);
261  float32 m_02 = xz - wy;
262  float32 m_01 = xy + wz;
263  float32 m_00 = 1.0f - (yy + zz);
264 
265  euler.x = std::atan2(m_12, m_22);
266  euler.y = std::asin(-m_02);
267  euler.z = std::atan2(m_01, m_00);
268 
269  return euler;
270 }

313 inline void Quaternion::Mul(const Quaternion* q2, Quaternion* res) const
314 {
315  const Quaternion* q1 = this;
316  float32 A, B, C, D, E, F, G, H;
317 
318  A = (q1->w + q1->x) * (q2->w + q2->x);
319  B = (q1->z - q1->y) * (q2->y - q2->z);
320  C = (q1->x - q1->w) * (q2->y + q2->z);
321  D = (q1->y + q1->z) * (q2->x - q2->w);
322  E = (q1->x + q1->z) * (q2->x + q2->y);
323  F = (q1->x - q1->z) * (q2->x - q2->y);
324  G = (q1->w + q1->y) * (q2->w - q2->z);
325  H = (q1->w - q1->y) * (q2->w + q2->z);
326 
327  res->w = B + (-E - F + G + H) * 0.5f;
328  res->x = A - (E + F + G + H) * 0.5f;
329  res->y = -C + (E - F + G - H) * 0.5f;
330  res->z = -D + (E - F - G + H) * 0.5f;
331 }

447 inline void Quaternion::Construct(const Vector3& euler)
448 {
449  {
450  float32 cy = std::cos(euler.z * 0.5f);
451  float32 sy = std::sin(euler.z * 0.5f);
452  float32 cr = std::cos(euler.x * 0.5f);
453  float32 sr = std::sin(euler.x * 0.5f);
454  float32 cp = std::cos(euler.y * 0.5f);
455  float32 sp = std::sin(euler.y * 0.5f);
456 
457  w = cy * cr * cp + sy * sr * sp;
458  x = cy * sr * cp - sy * cr * sp;
459  y = cy * cr * sp + sy * sr * cp;
460  z = sy * cr * cp - cy * sr * sp;

464 inline void Quaternion::Construct(const Matrix4& matrix)
465 {
466  //http://www.euclideanspace.com/maths/geometry/rotations/conversions/matrixToQuaternion/
467 
468  using mtx_elm = float32[4][4];
469  const mtx_elm& m = matrix._data;
470 
471  float32 tr, s, q[4];
472  int32 i, j, k;
473 
474  int32 nxt[3] = { 1, 2, 0 };
475 
476  tr = m[0][0] + m[1][1] + m[2][2];
477 
478  if (tr > 0.0f)
479  {
480  s = std::sqrt(tr + 1.0f);
481  w = s / 2.0f;
482  s = 0.5f / s;
483  x = (m[1][2] - m[2][1]) * s;
484  y = (m[2][0] - m[0][2]) * s;
485  z = (m[0][1] - m[1][0]) * s;
486  }
487  else
488  {
489  i = 0;
490  if (m[1][1] > m[0][0])
491  i = 1;
492  if (m[2][2] > m[i][i])
493  i = 2;
494  j = nxt[i];
495  k = nxt[j];
496 
497  s = std::sqrt((m[i][i] - (m[j][j] + m[k][k])) + 1.0f);
498 
499  q[i] = s * 0.5f;
500 
501  if (s != 0.0f)
502  s = 0.5f / s;
503 
504  q[3] = (m[j][k] - m[k][j]) * s;
505  q[j] = (m[i][j] + m[j][i]) * s;
506  q[k] = (m[i][k] + m[k][i]) * s;
507 
508  x = q[0];
509  y = q[1];
510  z = q[2];
511  w = q[3];
512  }
