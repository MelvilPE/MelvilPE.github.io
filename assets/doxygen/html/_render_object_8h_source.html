467 inline void RenderObject::AddVisibilityStructureNode(uint32 nodeValue)
468 {
469  inVisibilityNodes[inVisibilityNodeCount++] = nodeValue;

474 inline void RenderObject::RemoveVisibilityStructureNode(uint32 nodeValue)
475 {
476  DVASSERT(inVisibilityNodeCount <= MAX_VISIBILITY_STRUCTURE_NODE_COUNT);
477 
478  for (uint32 k = 0; k < inVisibilityNodeCount; ++k)
479  {
480  if (inVisibilityNodes[k] == nodeValue)
481  {
482  inVisibilityNodes[k] = inVisibilityNodes[inVisibilityNodeCount - 1];
483  inVisCopy[k] = inVisCopy[inVisibilityNodeCount - 1];
484  k--;
485 
486  DVASSERT(inVisibilityNodeCount > 0);
487  inVisibilityNodeCount--;
488  }
489  }
490 }
