22  static int32 const maxN = 0x477FE000; // max flt16 normal as a flt32
23  static int32 const minN = 0x38800000; // min flt16 normal as a flt32
24  static int32 const signN = 0x80000000; // flt32 sign bit

26  static int32 const infC = infN >> shift;
27  static int32 const nanN = (infC + 1) << shift; // minimum flt16 nan as a flt32
28  static int32 const maxC = maxN >> shift;
29  static int32 const minC = minN >> shift;
30  static int32 const signC = static_cast<uint32>(signN) >> shiftSign; // flt16 sign bit
31 
32  static int32 const mulN = 0x52000000; // (1 << 23) / minN
33  static int32 const mulC = 0x33800000; // minN / (1 << (23 - shift))
34 
35  static int32 const subC = 0x003FF; // max flt32 subnormal down shifted
36  static int32 const norC = 0x00400; // min flt32 normal down shifted
37 
38  static int32 const maxD = infC - maxC - 1;
39  static int32 const minD = minC - subC - 1;

41 public:
42  static uint16 Compress(float32 value)
43  {
44  Bits v, s;
45  v.f = value;
46  uint32_t sign = v.si & signN;
47  v.si ^= sign;
48  sign >>= shiftSign; // logical shift
49  s.si = mulN;
50  s.si = static_cast<int32>(s.f * v.f); // correct subnormals
51  v.si ^= (s.si ^ v.si) & -(minN > v.si);
52  v.si ^= (infN ^ v.si) & -((infN > v.si) & (v.si > maxN));
53  v.si ^= (nanN ^ v.si) & -((nanN > v.si) & (v.si > infN));
54  v.ui >>= shift; // logical shift
55  v.si ^= ((v.si - maxD) ^ v.si) & -(v.si > maxC);
56  v.si ^= ((v.si - minD) ^ v.si) & -(v.si > subC);
57  return v.ui | sign;
58  }

60  static float32 Decompress(uint16 value)
61  {
62  Bits v;
63  v.ui = value;
64  int32 sign = v.si & signC;
65  v.si ^= sign;
66  sign <<= shiftSign;
67  v.si ^= ((v.si + minD) ^ v.si) & -(v.si > subC);
68  v.si ^= ((v.si + maxD) ^ v.si) & -(v.si > maxC);
69  Bits s;
70  s.si = mulC;
71  s.f *= v.si;
72  int32 mask = -(norC > v.si);
73  v.si <<= shift;
74  v.si ^= (s.si ^ v.si) & mask;
75  v.si |= sign;
76  return v.f;
77  }
