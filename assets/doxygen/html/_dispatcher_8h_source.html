140 void Dispatcher<T>::SendEvent(U&& e, eSendPolicy policy)
141 {
142  DVASSERT(linkedThreadId != 0, "Before calling SendEvent you must call LinkToCurrentThread");
143 
144  uint64 curThreadId = Thread::GetCurrentIdAsUInt64();
145  if (linkedThreadId == curThreadId)
146  {
147  switch (policy)
148  {
149  case eSendPolicy::QUEUED_EXECUTION:
150  // clang-format off
151  {
152  LockGuard<Mutex> lock(mutex);
153  eventQueue.emplace_back(std::forward<U>(e));
154  }
155  // clang-format on
156  ProcessEvents();
157  break;
158  case eSendPolicy::IMMEDIATE_EXECUTION:
159  eventHandler(e);
160  break;
161  default:
162  DVASSERT(0);
163  break;
164  }
165  }
166  else
167  {
168  // Wait till current blocking call completion if any
169  semaphore.Wait();
170 
171  {
172  LockGuard<Mutex> lock(mutex);
173  eventQueue.emplace_back(std::forward<U>(e));
174  blockingCallInQueue = true;
175  }
176 
177  DAVA_BEGIN_BLOCKING_CALL(linkedThreadId);
178 
179  sendEventTrigger();
180 
181  signalEvent.Wait();
182  semaphore.Post(1);
183 
184  DAVA_END_BLOCKING_CALL(linkedThreadId);
185  }

188 template <typename T>
189 void Dispatcher<T>::ProcessEvents()
190 {
191  DVASSERT(!processEventsInProgress, "Dispatcher: nested call of ProcessEvents detected");
192 
193  bool shouldCompleteBlockingCall = false;
194  processEventsInProgress = true;
195  {
196  LockGuard<Mutex> lock(mutex);
197  eventQueue.swap(readyEvents);
198  std::swap(blockingCallInQueue, shouldCompleteBlockingCall);
199  }
200 
201  curEventIndex = 0;
202  for (const T& w : readyEvents)
203  {
204  eventHandler(w);
205  curEventIndex += 1;
206  }
207  readyEvents.clear();
208  if (shouldCompleteBlockingCall)
209  {
210  signalEvent.Signal();
211  }
212  processEventsInProgress = false;
213 }
