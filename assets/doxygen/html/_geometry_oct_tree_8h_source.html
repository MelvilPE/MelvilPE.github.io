121 inline AABBox3 GeometryOctTree::GetChildBox(const AABBox3& parentBox, uint32 childNodeIndex) const
122 {
123  Vector3 boundingBoxCenter = parentBox.min + parentBox.GetSize() / 2.0f;
124 
125  uint32 xdiv = (childNodeIndex >> 2) & 1;
126  uint32 ydiv = (childNodeIndex >> 1) & 1;
127  uint32 zdiv = (childNodeIndex)&1;
128 
129  Vector3 childBoxMin(
130  (xdiv == 0) ? (parentBox.min.x) : (boundingBoxCenter.x),
131  (ydiv == 0) ? (parentBox.min.y) : (boundingBoxCenter.y),
132  (zdiv == 0) ? (parentBox.min.z) : (boundingBoxCenter.z));
133 
134  Vector3 childBoxMax(
135  (xdiv == 0) ? (boundingBoxCenter.x) : (parentBox.max.x),
136  (ydiv == 0) ? (boundingBoxCenter.y) : (parentBox.max.y),
137  (zdiv == 0) ? (boundingBoxCenter.z) : (parentBox.max.z));
138 
139  return AABBox3(childBoxMin, childBoxMax);
140 }
