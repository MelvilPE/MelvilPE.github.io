106 template <class T>
107 inline bool BufferAllocator<T>::alloc(uint32 size, BufferAllocator<T>::Block* block)
108 {
109  bool success = false;
110 
111  void* mem = nullptr;
112  uint32 alignedSize = L_ALIGNED_SIZE(size, _granularity);
113 
114  DAVA::LockGuard<DAVA::Mutex> lock(pageLock);
115 
116  for (page_t& page : _page)
117  {
118  mem = page.heap->alloc(alignedSize);
119 
120  if (mem)
121  {
122  block->uid = page.bufferUid;
123  block->base = static_cast<uint32>(reinterpret_cast<uint8*>(mem) - MemBase);
124  block->size = size;
125  success = true;
126  break;
127  }
128  }
129 
130  if (mem == nullptr)
131  {
132  uint32 requiredSize = _page_sz;
133  while (requiredSize < alignedSize)
134  requiredSize *= 2;
135 
136  page_t page;
137  page.size = requiredSize;
138  page.heap = new SimpleRemoteHeap(8 * 1024);
139  page.heap->initialize(MemBase, page.size);
140  page.bufferUid = T::Create(requiredSize);
141  _page.push_back(page);
142 
143  mem = page.heap->alloc(alignedSize);
144 
145  if (mem)
146  {
147  block->uid = page.bufferUid;
148  block->base = static_cast<uint32>(reinterpret_cast<uint8*>(mem) - MemBase);
149  block->size = size;
150  success = true;
151  }
152  }
153 
154  return success;
155 }

158 inline bool BufferAllocator<T>::free(const BufferAllocator<T>::Block& block)
159 {
160  bool success = false;
161 
162  DAVA::LockGuard<DAVA::Mutex> lock(pageLock);
163 
164  for (auto p = _page.begin(), p_end = _page.end(); p != p_end; ++p)
165  {
166  if (block.uid == p->bufferUid)
167  {
168  p->heap->free(MemBase + block.base);
169 
170  if (p->heap->Empty())
171  {
172  delete p->heap;
173  T::Delete(p->bufferUid);
174  _page.erase(p);
175  }
176 
177  success = true;
178  break;
179  }
180  }
181 
182  return success;
