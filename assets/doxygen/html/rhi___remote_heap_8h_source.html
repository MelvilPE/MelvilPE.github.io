144 
145  Block* block = nullptr;

198 inline void SimpleRemoteHeap::_defragment()
199 {
200  bool has_free_blk = false;
201  bool do_wipe = true;
202 
203  for (const Block& b : _block)
204  {
205  if (b.IsAllocated())
206  {
207  do_wipe = false;
208  break;
209  }
210  }
211 
212  if (do_wipe)
213  {
214  _block.clear();
215  _unused = _base;
216  return;
217  }
218 
219  do
220  {
221  uint8_t* max_ptr = _base;
222  auto blk = _block.end();
223 
224  for (auto b = _block.begin(), b_end = _block.end(); b != b_end; ++b)
225  {
226  if (b->base > max_ptr)
227  {
228  blk = b;
229  max_ptr = b->base;
230  }
231  }
232 
233  if ((blk != _block.end()) && !blk->IsAllocated())
234  {
235  // de-commit block's memory
236  _unused -= blk->size;
237  _block.erase(blk);
238  has_free_blk = true;
239  }
240  else
241  {
242  has_free_blk = false;
243  }
244  }
245  while (has_free_blk);
