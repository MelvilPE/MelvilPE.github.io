185 template <class T, ResourceType RT, class DT, bool nr>
186 inline Handle ResourcePool<T, RT, DT, nr>::Alloc()
187 {
188  DAVA::LockGuard<DAVA::Spinlock> lock(ObjectSync);
189 
190  uint32 handle = InvalidHandle;
191  if (!Object)
192  {
193  DAVA_MEMORY_PROFILER_ALLOC_SCOPE(DAVA::ALLOC_POOL_RHI_RESOURCE_POOL);
194  Object = new Entry[ObjectCount];
195 
196  uint32 objectIndex = 0;
197  while (objectIndex < ObjectCount)
198  {
199  Entry& e = Object[objectIndex];
200  e.allocated = false;
201  e.generation = 0;
202 
203  ++objectIndex;
204  e.nextObjectIndex = objectIndex;
205  }
206 
207  (Object + ObjectCount - 1)->nextObjectIndex = 0;
208  }
209 
210  Entry* e = Object + HeadIndex;
211  RHI_POOL_ASSERT(!e->allocated, DAVA::Format("[RHIPool] Failed to allocate handle: pool is empty | Pool<%d>", RT).c_str());
212  HeadIndex = e->nextObjectIndex;
213 
214  e->allocated = true;
215  ++e->generation;
216 
217 #if (RHI_RESOURCE_INCLUDE_BACKTRACE)
218  e->CaptureBacktrace();
219 #endif
220 
221  handle = 0;
222  handle = ((uint32(e - Object) << HANDLE_INDEX_SHIFT) & HANDLE_INDEX_MASK) |
223  (((e->generation) << HANDLE_GENERATION_SHIFT) & HANDLE_GENERATION_MASK) |
224  ((RT << HANDLE_TYPE_SHIFT) & HANDLE_TYPE_MASK);
225 
226  return handle;
227 }
