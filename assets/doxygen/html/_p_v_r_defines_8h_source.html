111 /*!***************************************************************************

113  @Returns True if the platform the code is ran on is little endian
114  @Description Returns true if the platform the code is ran on is little endian
115  *****************************************************************************/
116 inline bool PVRTIsLittleEndian()
117 {
118  static bool bLittleEndian;
119  static bool bIsInit = false;
120 
121  if (!bIsInit)
122  {
123  short int word = 0x0001;
124  char* byte = reinterpret_cast<char*>(&word);
125  bLittleEndian = byte[0] ? true : false;
126  bIsInit = true;
127  }
128 
129  return bLittleEndian;
130 }
